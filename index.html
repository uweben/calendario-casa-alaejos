if (!cal) { return jsonResponse({ ok: false, error: 'Calendario no encontrado' }); } const now = new Date(); const start = new Date(now.getFullYear() - 1, 0, 1); const end = new Date(now.getFullYear() + 2, 11, 31, 23, 59, 59); const events = cal.getEvents(start, end); const items = events.map(ev => { const startDate = ev.getStartTime(); const endDate = ev.getEndTime(); const isAllDay = ev.isAllDayEvent(); const title = ev.getTitle() || ''; const description = ev.getDescription() || ''; const isCancelled = /^\[CANCELADO\]/im.test(title.trim()) || /^Estado:\s*CANCELADO/im.test(description); return { id: ev.getId(), title: title, description: description, isAllDay: isAllDay, cancelled: isCancelled, start: formatDateISO(startDate, isAllDay), end: formatDateISO(endDate, isAllDay) }; }); return jsonResponse({ ok: true, events: items }); } catch (err) { return jsonResponse({ ok: false, error: err.message }); } } function doPost(e) { try { const cal = CalendarApp.getCalendarById(CALENDAR_ID); if (!cal) { return jsonResponse({ ok: false, error: 'Calendario no encontrado' }); } const raw = e.postData && e.postData.contents ? e.postData.contents : '{}'; const data = JSON.parse(raw); if (data.action === 'cancel') { const eventId = (data.id || '').trim(); if (!eventId) { return jsonResponse({ ok: false, error: 'Falta el ID del evento a cancelar' }); } const ev = cal.getEventById(eventId); if (!ev) { return jsonResponse({ ok: false, error: 'Evento no encontrado en el calendario' }); } const currentTitle = ev.getTitle() || ''; if (!/^\[CANCELADO\]/i.test(currentTitle.trim())) { ev.setTitle(`[CANCELADO] ${currentTitle}`); } const currentDesc = ev.getDescription() || ''; if (!/^Estado:\s*CANCELADO/im.test(currentDesc)) { const stamp = Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd HH:mm'); const nextDesc = currentDesc ? `${currentDesc}\nEstado: CANCELADO (${stamp})` : `Estado: CANCELADO (${stamp})`; ev.setDescription(nextDesc); } return jsonResponse({ ok: true, id: eventId, cancelled: true }); } if (data.action === 'update') { const eventId = (data.id || '').trim(); if (!eventId) { return jsonResponse({ ok: false, error: 'Falta el ID del evento a editar' }); } const ev = cal.getEventById(eventId); if (!ev) { return jsonResponse({ ok: false, error: 'Evento no encontrado en el calendario' }); } const who = (data.who || '').trim(); const family = (data.family || '').trim(); const color = (data.color || '').trim(); const comment = (data.comment || '').trim(); const checkInStr = (data.checkIn || '').trim(); const checkOutStr = (data.checkOut || '').trim(); const isCancelled = Boolean(data.cancelled); if (!who || !checkInStr || !checkOutStr) { return jsonResponse({ ok: false, error: 'Faltan datos obligatorios para actualizar' }); } const startDay = parseLocalDate(checkInStr); const endDayExclusive = addDays(parseLocalDate(checkOutStr), 1); if (endDayExclusive <= startDay) { return jsonResponse({ ok: false, error: 'La fecha de salida debe ser igual o posterior a la de entrada' }); } const baseTitle = `Casa Alaejos: ${who}`; const nextTitle = isCancelled ? `[CANCELADO] ${baseTitle}` : baseTitle; const descLines = [ `Reservado por: ${who}`, `Familia: ${family}`, `Color: ${color}`, `Entrada: ${checkInStr}`, `Salida: ${checkOutStr}` ]; if (comment) { descLines.push(`Comentario: ${comment}`); } if (isCancelled) { const stamp = Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd HH:mm'); descLines.push(`Estado: CANCELADO (${stamp})`); } ev.setTitle(nextTitle); ev.setAllDayDates(startDay, endDayExclusive); ev.setDescription(descLines.join('\n')); return jsonResponse({ ok: true, id: eventId, updated: true }); } const who = (data.who || '').trim(); const family = (data.family || '').trim(); const color = (data.color || '').trim(); const comment = (data.comment || '').trim(); const checkInStr = (data.checkIn || '').trim(); const checkOutStr = (data.checkOut || '').trim(); if (!who || !checkInStr || !checkOutStr) { return jsonResponse({ ok: false, error: 'Faltan datos obligatorios (quién va, entrada o salida)' }); } const startDay = parseLocalDate(checkInStr); const endDayExclusive = addDays(parseLocalDate(checkOutStr), 1); if (endDayExclusive <= startDay) { return jsonResponse({ ok: false, error: 'La fecha de salida debe ser igual o posterior a la de entrada' }); } const title = `Casa Alaejos: ${who}`; const descLines = [ `Reservado por: ${who}`, `Familia: ${family}`, `Color: ${color}`, `Entrada: ${checkInStr}`, `Salida: ${checkOutStr}` ]; if (comment) { descLines.push(`Comentario: ${comment}`); } const created = cal.createAllDayEvent(title, startDay, endDayExclusive, { description: descLines.join('\n') }); return jsonResponse({ ok: true, id: created.getId(), title: title, start: checkInStr, end: checkOutStr }); } catch (err) { return jsonResponse({ ok: false, error: err.message }); } } function parseLocalDate(yyyyMmDd) { const parts = yyyyMmDd.split('-').map(Number); return new Date(parts[0], parts[1] - 1, parts[2], 0, 0, 0); } function addDays(dateObj, days) { return new Date(dateObj.getFullYear(), dateObj.getMonth(), dateObj.getDate() + days, 0, 0, 0); } function formatDateISO(dateObj, isAllDay) { if (isAllDay) { return Utilities.formatDate(dateObj, TIMEZONE, 'yyyy-MM-dd'); } return Utilities.formatDate(dateObj, TIMEZONE, "yyyy-MM-dd'T'HH:mm:ssXXX"); } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
